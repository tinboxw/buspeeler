cmake_minimum_required(VERSION 3.16)
project(buspeeler_collector LANGUAGES CXX)
add_executable(buspeeler-collector main.cpp)
target_compile_features(buspeeler-collector PRIVATE cxx_std_17)
find_package(Threads REQUIRED)
target_link_libraries(buspeeler-collector PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(buspeeler-collector PRIVATE /W4 /utf-8)
else()
  target_compile_options(buspeeler-collector PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(buspeeler-collector PRIVATE ${CMAKE_DL_LIBS})
endif()
if(MINGW)
  target_link_options(buspeeler-collector PRIVATE -static-libgcc -static-libstdc++ -static)
endif()
